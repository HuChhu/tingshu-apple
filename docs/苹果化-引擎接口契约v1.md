// commonMain —— 引擎侧全部纯函数 + suspend，零平台 import
interface SourceActions {
    val sourceId: String      // UUID，全局唯一（§19.1）

    val capabilities: Capabilities

interface AudioExtractor {                                                  // C2：去命令化
    suspend fun extract(url: String, timeoutMs: Int = 15_000): ExtractedAudio   // 纯返回，autoPlay/isCache/isDebug 留壳侧

data class ExtractedAudio(val url: String, val headers: Map<String, String>)    // headers 按 host 精确匹配才注入（§19.2）

    data class NotFound(val url: String) : SourceErrors
    data class ParseFailed(val reason: String) : SourceErrors
    data class RateLimited(val needVerification: Boolean) : SourceErrors    // 对齐 bridge「不硬冲」语义
    data class Network(val cause: String) : SourceErrors
    data class Auth(val reason: String) : SourceErrors
    data class Upstream(val status: Int) : SourceErrors
}
interface PlaybackStateSink { fun onState(state: PlaybackState) }           // enum { PLAYING, PAUSED, ERROR, STOPPED }

## actual 触点收敛（C3：shared/actual/ 文件数 ≤5）
WebViewSniffer.actual / WebViewContainer.actual（ILogin+验证共用）/ CookieStore.actual / Player.actual（AVPlayer）/ BackgroundSession.actual
解析、限速、错误分类、签名引擎全部 commonMain。
## 签名引擎同步纪律
同一算法两处实现：commonMain 引擎 ↔ ximalaya-bridge upstream.py——改一处必须同步另一处，写进每次提交检查。
## 行为基线
KMP 核心行为 = 桥的现有语义（/api/capabilities 能力声明、降级 hint、播放地址 expiresInSec）；验收时双端同输入对拍输出。
## 书源 JSON schema 边界（法律席）
只准数据字段：URL 模板 / CSS·XPath 选择器 / JSONPath / 声明式参数；schema 校验禁一切可执行脚本字段。

## v1.1 修订（2026-11-21，dev-core 提案，lead 批准，M2 实现已采用）
1. SourceErrors.Network 字段 cause → reason（与全族字段名一致）
2. sealed interface SourceErrors → sealed class（Kotlin 2.0.21 可抛化需要）
3. +CookieProvider SPI：suspend fun cookie(): String? / suspend fun refresh()——凭据来源按平台注入（JVM=桥凭据 env 路径；iOS M4=WKWebView 官方页登录显式拷贝）
> 尾部扩展（Episode.coverUrl / ExtractedAudio.expiresInSeconds）一并收录。架构席如需复核，差异均为机械性，不影响 C1/C2/C3 边界。

## v1.1 使用纪律（架构席终审注记 2026-11-21，非阻塞两条）
1. sealed class 可抛化 ≠ 允许异常控制流：引擎层只返回 SourceErrors（对拍/分类走 ErrorClassifier 纯函数），throw 仅限壳侧崩溃上报——桥的「失败不重试」纪律建立在返回值分类上，不得变成 try/catch 链。
2. CookieProvider.refresh() 的 iOS 实现必须复用 WebViewContainer.actual 完成登录拷贝，禁止新增第六个 actual——actual ≤5 不变量由此闭环。
> 架构席终审结论：三项修订无异议，v1.1 即终版。
